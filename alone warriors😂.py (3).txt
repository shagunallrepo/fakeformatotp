# CopyTextButton exists in python-telegram-bot >= 21.11.
try:
    from telegram import CopyTextButton  # type: ignore
    _COPY_SUPPORTED = True
except ImportError:
    class CopyTextButton:
        def __init__(self, text: str):
            self.text = text
    _COPY_SUPPORTED = False

# Button style ("danger" / "primary" / "success") exists in PTB >= 21.11 too.
try:
    from telegram import InlineKeyboardButton as _IKB  # noqa
    _STYLE_SUPPORTED = "style" in _IKB.__init__.__code__.co_varnames
except Exception:
    _STYLE_SUPPORTED = False

def _btn(label: str, *, url: Optional[str] = None,
         copy_text: Optional[CopyTextButton] = None,
         style: Optional[str] = None) -> InlineKeyboardButton:
    """Build an InlineKeyboardButton, applying style only if supported."""
    kwargs = {"text": label}
    if url is not None:
        kwargs["url"] = url
    if copy_text is not None:
        kwargs["copy_text"] = copy_text
    if style is not None and _STYLE_SUPPORTED:
        kwargs["style"] = style
    return InlineKeyboardButton(**kwargs)

def setup_logging() -> logging.Logger:
    logger = logging.getLogger("otpbot")
    logger.setLevel(logging.INFO)
    fmt = logging.Formatter(
        "%(asctime)s | %(levelname)-7s | %(name)s | %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
    )
    fh = logging.handlers.RotatingFileHandler(
        LOG_PATH, maxBytes=5 * 1024 * 1024, backupCount=5, encoding="utf-8"
    )
    fh.setFormatter(fmt)
    sh = logging.StreamHandler()
    sh.setFormatter(fmt)
    logger.addHandler(fh)
    logger.addHandler(sh)
    return logger

log = setup_logging()

def db_init() -> None:
    with sqlite3.connect(DB_PATH) as con:
        con.executescript(
            """
            CREATE TABLE IF NOT EXISTS settings (
                key   TEXT PRIMARY KEY,
                value TEXT NOT NULL
            );
            CREATE TABLE IF NOT EXISTS admins (
                user_id INTEGER PRIMARY KEY,
                added_at INTEGER NOT NULL
            );
            CREATE TABLE IF NOT EXISTS templates (
                country TEXT PRIMARY KEY,
                body    TEXT NOT NULL,
                updated INTEGER NOT NULL
            );
            CREATE TABLE IF NOT EXISTS stats (
                day     TEXT NOT NULL,
                country TEXT NOT NULL,
                count   INTEGER NOT NULL DEFAULT 0,
                PRIMARY KEY (day, country)
            );
            CREATE TABLE IF NOT EXISTS audit (
                ts      INTEGER NOT NULL,
                user_id INTEGER NOT NULL,
                action  TEXT NOT NULL,
                detail  TEXT
            );
            """
        )
        defaults = {
            "group_id":     str(DEFAULT_GROUP_ID),
            "interval_min": "5",
            "interval_max": "12",
            "autopost":     "0",
            "country":      "VE",
            "numbers_url":  NUMBERS_URL,
            "chat_url":     CHAT_URL,
            "bot_url":      BOT_URL,
            "active_from":  "00:00",
            "active_to":    "23:59",
            "pinned_note":  "",
            "prefix":       "KITE",
            "counter":      "0",
            "dry_run":      "0",
        }
        for k, v in defaults.items():
            con.execute("INSERT OR IGNORE INTO settings(key, value) VALUES (?, ?)", (k, v))
        con.commit()
    log.info("db initialised at %s", DB_PATH)

def get_setting(key: str, default: str = "") -> str:
    with sqlite3.connect(DB_PATH) as con:
        row = con.execute("SELECT value FROM settings WHERE key = ?", (key,)).fetchone()
        return row[0] if row else default

def set_setting(key: str, value: str) -> None:
    with sqlite3.connect(DB_PATH) as con:
        con.execute(
            "INSERT INTO settings(key, value) VALUES (?, ?) "
            "ON CONFLICT(key) DO UPDATE SET value = excluded.value",
            (key, str(value)),
        )
        con.commit()

def bump_counter(country: str) -> int:
    today = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    with sqlite3.connect(DB_PATH) as con:
        con.execute(
            "INSERT INTO stats(day, country, count) VALUES (?, ?, 1) "
            "ON CONFLICT(day, country) DO UPDATE SET count = count + 1",
            (today, country),
        )
        con.execute("UPDATE settings SET value = CAST(value AS INTEGER) + 1 WHERE key = 'counter'")
        con.commit()
        row = con.execute("SELECT value FROM settings WHERE key = 'counter'").fetchone()
        return int(row[0]) if row else 0

def get_stats(days: int = 7) -> dict:
    cutoff = (datetime.now(timezone.utc) - timedelta(days=days)).strftime("%Y-%m-%d")
    with sqlite3.connect(DB_PATH) as con:
        rows = con.execute(
            "SELECT day, country, count FROM stats WHERE day >= ? ORDER BY day DESC",
            (cutoff,),
        ).fetchall()
    return {"rows": rows}

def audit(user_id: int, action: str, detail: str = "") -> None:
    with sqlite3.connect(DB_PATH) as con:
        con.execute(
            "INSERT INTO audit(ts, user_id, action, detail) VALUES (?, ?, ?, ?)",
            (int(time.time()), user_id, action, detail),
        )
        con.commit()

def is_admin(user_id: int) -> bool:
    if user_id == OWNER_ID:
        return True
    with sqlite3.connect(DB_PATH) as con:
        row = con.execute("SELECT 1 FROM admins WHERE user_id = ?", (user_id,)).fetchone()
    return row is not None

def add_admin(user_id: int) -> None:
    with sqlite3.connect(DB_PATH) as con:
        con.execute(
            "INSERT OR IGNORE INTO admins(user_id, added_at) VALUES (?, ?)",
            (user_id, int(time.time())),
        )
        con.commit()

def remove_admin(user_id: int) -> None:
    with sqlite3.connect(DB_PATH) as con:
        con.execute("DELETE FROM admins WHERE user_id = ?", (user_id,))
        con.commit()

def list_admins() -> list:
    with sqlite3.connect(DB_PATH) as con:
        rows = con.execute("SELECT user_id FROM admins ORDER BY added_at").fetchall()
    return [r[0] for r in rows]

# TEMPLATE ENGINE
# ═════════════════════════════════════════════════════════════
DEFAULT_TEMPLATE = """\
🌍 ┃ 📱
#{{ code_bold }}  {{ mid_bold }}-{{ tail_bold }}  🌐#{{ lang_tag }}"""

JINJA = Environment(loader=BaseLoader(), autoescape=False)

def get_template(country: str) -> str:
    with sqlite3.connect(DB_PATH) as con:
        row = con.execute("SELECT body FROM templates WHERE country = ?", (country,)).fetchone()
    return row[0] if row else DEFAULT_TEMPLATE

def set_template(country: str, body: str) -> None:
    with sqlite3.connect(DB_PATH) as con:
        con.execute(
            "INSERT INTO templates(country, body, updated) VALUES (?, ?, ?) "
            "ON CONFLICT(country) DO UPDATE SET body = excluded.body, updated = excluded.updated",
            (country, body, int(time.time())),
        )
        con.commit()

def render_otp(country_code: str):
    cc = COUNTRIES.get(country_code, COUNTRIES["VE"])
    digits    = "".join(random.choices(string.digits, k=6))
    secondary = "".join(random.choices(string.digits, k=5))
    prefix    = get_setting("prefix") or cc.default_prefix
    mid       = f"{prefix}-{digits}"
    tail      = secondary

    body_tmpl = get_template(cc.code)
    try:
        tmpl = JINJA.from_string(body_tmpl)
        text = tmpl.render(
            code_bold=bold(cc.code),
            mid_bold=bold(mid),
            tail_bold=bold(tail),
            lang_tag=cc.lang_tag,
            flag=cc.flag,
            country=cc.code,
            prefix=prefix,
            digits=digits,
            tail=tail,
        )
    except TemplateError as e:
        log.warning("template render failed for %s: %s", cc.code, e)
        text = f"#{bold(cc.code)}  {bold(mid)}-{bold(tail)}  🌐#{cc.lang_tag}"

    # The copy button carries a fresh 6-digit code (independent of display)
    copy_code = "".join(random.choices(string.digits, k=6))
    return text, copy_code

def build_keyboard(raw_code: str) -> InlineKeyboardMarkup:
    rows = []

    if SHOW_COPY_BUTTON and _COPY_SUPPORTED:
        rows.append([
            _btn(f"📋 {raw_code}", copy_text=CopyTextButton(text=raw_code), style="danger")
        ])

    rows.append([
        _btn("📞 Numbers", url=get_setting("numbers_url") or NUMBERS_URL, style="danger"),
        _btn("💬 Chat",    url=get_setting("chat_url")    or CHAT_URL,    style="primary"),
        _btn("🤖 Bot",     url=get_setting("bot_url")     or BOT_URL,     style="primary"),
    ])

    return InlineKeyboardMarkup(rows)

# ═════════════════════════════════════════════════════════════
# ACTIVE-HOURS WINDOW
# ═════════════════════════════════════════════════════════════
def within_active_window() -> bool:
    try:
        now = datetime.now(timezone.utc).time()
        frm = datetime.strptime(get_setting("active_from", "00:00"), "%H:%M").time()
        to  = datetime.strptime(get_setting("active_to",   "23:59"), "%H:%M").time()
    except Exception:
        return True
    if frm <= to:
        return frm <= now <= to
    return now >= frm or now <= to

# ═════════════════════════════════════════════════════════════
# POSTING
# ═════════════════════════════════════════════════════════════
async def post_otp(app: Application, chat_id: int, force: bool = False) -> bool:
    if not force and not within_active_window():
        return False

    cc_code = get_setting("country")
    text, raw = render_otp(cc_code)
    kb = build_keyboard(raw)

    note = get_setting("pinned_note")
    if note:
        text = f"{text}\n\n📌 {note}"

    if DRY_RUN or get_setting("dry_run") == "1":
        log.info("[DRY-RUN] would post to %s: %s", chat_id, text.replace("\n", " | "))
        return True

    try:
        await app.bot.send_message(
            chat_id=chat_id,
            text=text,
            reply_markup=kb,
            parse_mode=ParseMode.HTML,
            disable_web_page_preview=True,
        )
        bump_counter(cc_code)
        log.info("posted OTP to %s country=%s", chat_id, cc_code)
        return True
    except RetryAfter as e:
        log.warning("rate limited, sleeping %ss", e.retry_after)
        await asyncio.sleep(e.retry_after + 1)
        return await post_otp(app, chat_id, force=force)
    except (Forbidden, BadRequest) as e:
        log.error("send failed to %s: %s", chat_id, e)
        return False
    except TelegramError as e:
        log.error("telegram error: %s", e)
        return False

async def autopost_loop(app: Application) -> None:
    while True:
        try:
            if get_setting("autopost") == "1":
                try:
                    lo = max(1, int(get_setting("interval_min", "5")))
                    hi = max(lo, int(get_setting("interval_max", "12")))
                except ValueError:
                    lo, hi = 5, 12

                try:
                    gid = int(get_setting("group_id"))
                    await post_otp(app, gid)
                except Exception as e:
                    log.exception("post failed: %s", e)

                await asyncio.sleep(random.uniform(lo, hi))
            else:
                await asyncio.sleep(1)
        except asyncio.CancelledError:
            break
        except Exception:
            log.exception("autopost loop error")
            await asyncio.sleep(2)

# AUTH DECORATOR
# ═════════════════════════════════════════════════════════════
def admin_only(func):
    async def wrapper(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
        uid = update.effective_user.id if update.effective_user else 0
        if not is_admin(uid):
            if update.effective_message:
                try:
                    await update.effective_message.delete()
                except Exception:
                    pass
            return
        return await func(update, ctx)
    return wrapper

# ═════════════════════════════════════════════════════════════
# COUNTRY PICKER (inline menu)
# ═════════════════════════════════════════════════════════════
def country_menu_kb() -> InlineKeyboardMarkup:
    current = get_setting("country")
    rows = []
    per_row = 3
    items = list(COUNTRIES.values())
    for i in range(0, len(items), per_row):
        chunk = items[i:i + per_row]
        row = []
        for c in chunk:
            mark = "✅ " if c.code == current else ""
            row.append(_btn(
                f"{mark}{c.flag} {c.code}",
                style="success" if c.code == current else None,
            ))
            # NOTE: buttons without url/copy have no payload unless we add callback_data.
            # We'll rebuild with callback_data below.
        rows.append(row)
    # Rebuild properly with callback_data (can't mix _btn kwargs here cleanly)
    rows = []
    for i in range(0, len(items), per_row):
        chunk = items[i:i + per_row]
        row = []
        for c in chunk:
            label = ("✅ " if c.code == current else "") + f"{c.flag} {c.code}"
            row.append(InlineKeyboardButton(label, callback_data=f"cc:{c.code}"))
        rows.append(row)
    rows.append([InlineKeyboardButton("⬅️ Close", callback_data="cc:__close__")])
    return InlineKeyboardMarkup(rows)

@admin_only
async def cmd_speed(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    if len(ctx.args) < 2 or not all(a.isdigit() for a in ctx.args[:2]):
        await update.message.reply_text("usage: /speed <min_sec> <max_sec>")
        return
    lo, hi = int(ctx.args[0]), int(ctx.args[1])
    if lo < 1 or hi < lo:
        await update.message.reply_text("invalid range")
        return
    set_setting("interval_min", str(lo))
    set_setting("interval_max", str(hi))
    await update.message.reply_text(f"⚡ cadence = {lo}–{hi}s (randomized)")

@admin_only
async def cmd_active(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    if len(ctx.args) < 2:
        await update.message.reply_text("usage: /active HH:MM HH:MM")
        return
    try:
        datetime.strptime(ctx.args[0], "%H:%M")
        datetime.strptime(ctx.args[1], "%H:%M")
    except ValueError:
        await update.message.reply_text("bad time format")
        return
    set_setting("active_from", ctx.args[0])
    set_setting("active_to",   ctx.args[1])
    await update.message.reply_text(f"🕒 active window = {ctx.args[0]} → {ctx.args[1]} (UTC)")

# CALLBACKS
# ═════════════════════════════════════════════════════════════
async def on_callback(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    q = update.callback_query
    if not q:
        return
    uid = q.from_user.id if q.from_user else 0
    if not is_admin(uid):
        try:
            await q.answer("not authorized", show_alert=False)
        except TelegramError:
            pass
        return

    data = q.data or ""

    if data == "cc:__close__":
        try:
            await q.message.delete()
        except TelegramError:
            pass
        try:
            await q.answer()
        except TelegramError:
            pass
        return

    if data.startswith("cc:"):
        cc = data[3:].upper()
        if cc in COUNTRIES:
            set_setting("country", cc)
            try:
                await q.answer(f"country → {cc} ({COUNTRIES[cc].label})")
            except TelegramError:
                pass
            try:
                await q.edit_message_text(
                    f"🌍 Selected: {COUNTRIES[cc].flag} <b>{cc}</b> — {COUNTRIES[cc].label}",
                    parse_mode=ParseMode.HTML,
                    reply_markup=country_menu_kb(),
                )
            except TelegramError:
                pass
        else:
            try:
                await q.answer("unknown code")
            except TelegramError:
                pass
        return

    try:
        await q.answer()
    except TelegramError:
        pass
